Mục tiêu đầu ra: Biết đếm số phép so sánh của một thuật toán, phân loại Big-O cơ bản O(1), O(n), O(n²); thiết kế bộ test (mảng rỗng, 1 phần tử, đã sắp, sắp ngược, trùng lặp) và tự phát hiện, chữa lỗi trong code Linear Search / Bubble Sort.

1. Lý thuyết cốt lõi (35 Phút)

1.1 Đo "độ nhanh" bằng cách đếm phép toán

Cùng một chương trình chạy trên máy mạnh và máy yếu sẽ cho thời gian khác nhau, nên lập trình viên không đo thuật toán bằng giây. Thay vào đó, ta đếm số phép toán cơ bản (thường là số lần so sánh) và xem con số đó tăng thế nào khi số phần tử n tăng. Cách mô tả tốc độ tăng này gọi là độ phức tạp thuật toán, ký hiệu Big-O.

1.2 Ba mức Big-O cơ bản

Big-O Tên gọi Dấu hiệu trong code Ví dụ đã học
O(1) Hằng số Không có vòng lặp phụ thuộc n Truy cập arr[i], đổi chỗ 2 phần tử
O(n) Tuyến tính Một vòng lặp chạy qua n phần tử Tính tổng, tìm Max/Min (Buổi 27), Linear Search (Buổi 28)
O(n²) Bình phương Hai vòng lặp lồng nhau, mỗi vòng ~n lần Bubble Sort (Buổi 29), xét mọi cặp phần tử

1.3 Số phép so sánh khi n tăng

n Linear Search xấu nhất (n) Bubble Sort xấu nhất n(n−1)/2 Bubble Sort tốt nhất (đã sắp, có dừng sớm): n − 1
1010459
1001004 95099
1 0001 000499 500999
10 00010 00049 995 0009 999

Vì sao Bubble Sort là n(n−1)/2? Lượt 1 so sánh n − 1 cặp, lượt 2 so sánh n − 2 cặp, ..., lượt cuối 1 cặp. Tổng (n − 1) + (n − 2) + ... + 1 = n(n − 1)/2 ≈ n²/2. Big-O bỏ qua hằng số 1/2, nên ta nói Bubble Sort là O(n²): n tăng 10 lần → công việc tăng khoảng 100 lần.

big_o_examples.cpp 
// O(1): so phep toan KHONG phu thuoc n
int getFirst(int arr[]) {
    return arr[0];
}

// O(n): mot vong lap qua n phan tu
int sumArray(int arr[], int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    return sum;
}

// O(n^2): xet moi cap (i, j) - hai vong lap long nhau
int countEqualPairs(int arr[], int n) {
    int count = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] == arr[j]) count++;
        }
    }
    return count;
}

📌 Lưu ý về "mảng rỗng": C++ chuẩn không cho khai báo int a[0]. Để test trường hợp rỗng, ta khai báo mảng bình thường (ví dụ int a[1]) rồi truyền n = 0 vào hàm — hàm đúng phải không đụng tới phần tử nào.
🎮 Mẹo tư duy Game: Game chạy 60 khung hình/giây, tức mỗi khung chỉ có khoảng 16 mili-giây. Nếu có 1 000 viên đạn và bạn kiểm tra va chạm mọi cặp đạn–đạn (O(n²)) thì mỗi khung phải xét ~500 000 cặp — game sẽ giật. Đó là lý do các engine dùng "lưới không gian" để chỉ kiểm tra những vật ở gần nhau. Biết đếm phép toán giúp bạn đoán trước chỗ nào sẽ làm game chậm.
⚠️ Bảng lỗi kinh điển cần soi khi chữa bài:
  • i <= n thay vì i < n → đọc ô arr[n] ngoài mảng (off-by-one).
  • else return -1; nằm trong vòng lặp tìm kiếm → chỉ kiểm tra đúng phần tử đầu tiên.
  • Vòng trong Bubble Sort chạy tới j < n - i → khi i = 0, arr[j + 1] chính là arr[n].
  • Hoán đổi không dùng biến tạm → hai ô cùng một giá trị, dữ liệu bị nhân bản.
  • Khởi tạo max = 0 → sai khi mọi phần tử đều âm; nên khởi tạo max = arr[0] (và kiểm tra n > 0 trước).

2. Bài tập trên lớp (45 Phút)

Bài 1: Đo Bubble Sort trên bộ test chuẩn

Trên lớp

Đề bài: Sửa hàm Bubble Sort (có dừng sớm) để trả về số phép so sánh đã thực hiện. Viết hàm isSorted kiểm tra mảng đã tăng dần chưa, và hàm runTest in kết quả PASS/FAIL. Chạy trên 6 trường hợp: rỗng, 1 phần tử, đã sắp, sắp ngược, ngẫu nhiên, có trùng lặp.

sort_test_suite.cpp 
#include <iostream>

// Bubble Sort tang dan, tra ve so lan so sanh
int bubbleSortCount(int arr[], int n) {
    int comparisons = 0;
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            comparisons++;                        // Dem moi lan so sanh
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = true;
            }
        }
        if (!swapped) break;
    }
    return comparisons;
}

// Kiem tra mang da tang dan chua - O(n)
bool isSorted(int arr[], int n) {
    for (int i = 0; i + 1 < n; i++) {
        if (arr[i] > arr[i + 1]) return false;
    }
    return true;
}

void runTest(int testId, int arr[], int n) {
    int comps = bubbleSortCount(arr, n);
    std::cout << "Test " << testId << " | n = " << n
              << " | so sanh: " << comps;
    if (isSorted(arr, n)) {
        std::cout << " | PASS\n";
    } else {
        std::cout << " | FAIL\n";
    }
}

int main() {
    int emptyArr[1] = {0};
    int single[1]   = {42};
    int sorted[6]   = {1, 2, 3, 4, 5, 6};
    int reversed[6] = {6, 5, 4, 3, 2, 1};
    int mixed[6]    = {4, 1, 6, 2, 5, 3};
    int dup[6]      = {3, 1, 3, 1, 2, 2};

    runTest(1, emptyArr, 0);    // Mang rong
    runTest(2, single, 1);      // 1 phan tu
    runTest(3, sorted, 6);      // Da sap - tot nhat
    runTest(4, reversed, 6);    // Sap nguoc - xau nhat
    runTest(5, mixed, 6);       // Ngau nhien
    runTest(6, dup, 6);         // Co trung lap
    return 0;
}

3. Bài tập về nhà (Tự luyện tập)

Mục tiêu tuần này: không chỉ viết code chạy được, mà phải chứng minh code đúng bằng bộ test và giải thích được độ phức tạp.

Bài 1 [Easy]: Thấy tận mắt O(n) và O(n²)

Easy (40%)

Tạo mảng int arr[1000] với arr[i] = i * 2. Viết hai hàm: (a) int countSearchSteps(int arr[], int n, int target) trả về số lần so sánh Linear Search thực hiện; (b) int countAllPairs(int n) đếm số cặp (i, j) với i < j bằng hai vòng lặp lồng nhau. In bảng kết quả với n = 10, 100, 1000 khi tìm target = -1 (không có trong mảng — trường hợp xấu nhất).

growth_demo.cpp 
#include <iostream>

// Dem so lan so sanh cua Linear Search - O(n)
int countSearchSteps(int arr[], int n, int target) {
    int steps = 0;
    for (int i = 0; i < n; i++) {
        steps++;
        if (arr[i] == target) break;
    }
    return steps;
}

// Dem so cap (i, j) voi i < j - O(n^2)
int countAllPairs(int n) {
    int pairs = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            pairs++;
        }
    }
    return pairs;
}

int main() {
    const int MAX = 1000;
    int arr[MAX];
    for (int i = 0; i < MAX; i++) {
        arr[i] = i * 2;
    }

    std::cout << "n\tO(n) search\tO(n^2) pairs\n";
    for (int n = 10; n <= MAX; n *= 10) {
        std::cout << n << "\t"
                  << countSearchSteps(arr, n, -1) << "\t\t"
                  << countAllPairs(n) << "\n";
    }
    return 0;
}

Kết quả: cột O(n) là 10, 100, 1000; cột O(n²) là 45, 4950, 499500. Khi n tăng 10 lần, cột đầu tăng 10 lần còn cột sau tăng ~100 lần.

Bài 2 [Medium]: Viết hàm tìm vị trí Max kèm bộ test tự động

Medium (40%)

Viết hàm int findMaxIndex(int arr[], int n) trả về chỉ số của phần tử lớn nhất (nếu có nhiều giá trị Max bằng nhau thì trả về chỉ số đầu tiên; mảng rỗng trả về -1). Viết hàm void check(int testId, int actual, int expected) in PASS/FAIL, và kiểm tra ít nhất 6 test: rỗng, 1 phần tử, Max ở đầu, Max ở cuối, toàn số âm, Max trùng lặp.

Phân tích: Khởi tạo maxIdx = 0 (không phải giá trị 0!) rồi so sánh bằng dấu > nghiêm ngặt để giữ chỉ số đầu tiên khi bằng nhau. Độ phức tạp O(n).

max_index_tests.cpp 
#include <iostream>

int findMaxIndex(int arr[], int n) {
    if (n <= 0) return -1;                     // Mang rong
    int maxIdx = 0;
    for (int i = 1; i < n; i++) {
        if (arr[i] > arr[maxIdx]) {               // Dau > giu chi so DAU TIEN
            maxIdx = i;
        }
    }
    return maxIdx;
}

void check(int testId, int actual, int expected) {
    std::cout << "Test " << testId << ": ";
    if (actual == expected) {
        std::cout << "PASS\n";
    } else {
        std::cout << "FAIL (nhan " << actual
                  << ", mong doi " << expected << ")\n";
    }
}

int main() {
    int emptyArr[1] = {0};
    int single[1]   = {7};
    int maxFirst[4] = {90, 10, 20, 30};
    int maxLast[4]  = {10, 20, 30, 90};
    int negative[4] = {-8, -3, -5, -9};
    int dupMax[5]   = {4, 9, 2, 9, 1};

    check(1, findMaxIndex(emptyArr, 0), -1);
    check(2, findMaxIndex(single, 1), 0);
    check(3, findMaxIndex(maxFirst, 4), 0);
    check(4, findMaxIndex(maxLast, 4), 3);
    check(5, findMaxIndex(negative, 4), 1);
    check(6, findMaxIndex(dupMax, 5), 1);
    return 0;
}

Bài 3 [Hard]: Đo "độ hỗn loạn" của bảng xếp hạng

Hard (20%)

Một cặp nghịch thế là cặp chỉ số i < j mà arr[i] > arr[j] (đứng sai thứ tự). Số cặp nghịch thế cho biết mảng "hỗn loạn" đến mức nào.

  • Viết int countInversions(int arr[], int n) bằng hai vòng lặp lồng nhau — O(n²).
  • Viết int bubbleSortSwaps(int arr[], int n) sắp xếp tăng dần và trả về số lần hoán đổi.
  • Trên bộ test (rỗng, 1 phần tử, đã sắp, sắp ngược, ngẫu nhiên, trùng lặp), chứng minh bằng chương trình rằng số lần hoán đổi của Bubble Sort luôn bằng số cặp nghịch thế, đồng thời mảng sau cùng đã được sắp xếp.
  • Trả lời: mảng n phần tử có tối đa bao nhiêu cặp nghịch thế? Rơi vào trường hợp nào?

Ý tưởng: Mỗi lần Bubble Sort đổi chỗ hai phần tử liền kề sai thứ tự, nó sửa đúng một cặp nghịch thế và không tạo thêm cặp mới. Vì vậy số lần đổi = số cặp nghịch thế ban đầu. Lưu ý phải đếm nghịch thế trước khi sắp xếp, vì sắp xong thì số nghịch thế bằng 0.

inversions.cpp 
#include <iostream>

// Dem cap (i < j) ma arr[i] > arr[j] - O(n^2)
int countInversions(int arr[], int n) {
    int count = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] > arr[j]) count++;
        }
    }
    return count;
}

// Bubble Sort tang dan, tra ve so lan hoan doi
int bubbleSortSwaps(int arr[], int n) {
    int swaps = 0;
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swaps++;
                swapped = true;
            }
        }
        if (!swapped) break;
    }
    return swaps;
}

bool isSorted(int arr[], int n) {
    for (int i = 0; i + 1 < n; i++) {
        if (arr[i] > arr[i + 1]) return false;
    }
    return true;
}

void runTest(int testId, int arr[], int n) {
    int inversions = countInversions(arr, n);   // Dem TRUOC khi sap xep
    int swaps = bubbleSortSwaps(arr, n);
    std::cout << "Test " << testId << " | nghich the: " << inversions
              << " | hoan doi: " << swaps;
    if (inversions == swaps && isSorted(arr, n)) {
        std::cout << " | PASS\n";
    } else {
        std::cout << " | FAIL\n";
    }
}

int main() {
    int emptyArr[1] = {0};
    int single[1]   = {42};
    int sorted[5]   = {100, 200, 300, 400, 500};
    int reversed[5] = {500, 400, 300, 200, 100};
    int mixed[5]    = {300, 100, 500, 200, 400};
    int dup[5]      = {200, 100, 200, 100, 200};

    runTest(1, emptyArr, 0);
    runTest(2, single, 1);
    runTest(3, sorted, 5);
    runTest(4, reversed, 5);
    runTest(5, mixed, 5);
    runTest(6, dup, 5);
    return 0;
}
💡 Trả lời câu hỏi cuối: Tối đa là n(n − 1)/2 cặp nghịch thế, khi mảng sắp ngược hoàn toàn (mọi cặp đều sai thứ tự). Với 5 phần tử: 5·4/2 = 10 — đúng bằng số hoán đổi ở Test 4. Đây cũng là lý do Bubble Sort xấu nhất là O(n²): nó chỉ sửa được một cặp sai mỗi lần đổi chỗ.
